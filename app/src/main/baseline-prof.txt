# Ahead-of-time compile Exhale's own code at install time (profileinstaller applies this).
#
# Without a profile, every screen, spring and gesture runs interpreted/JIT-compiled the first few
# times it's used — that's the stutter on a cold start, the first open of the player, the first
# swipe of the dock. Compose and AndroidX already ship their own profiles; this covers the app and
# the glass libraries it draws with. Costs some install time and a few MB of storage.
HSPLcom/ozyern/exhale/**->**(**)**
Lcom/ozyern/exhale/**;
HSPLcom/kyant/**->**(**)**
Lcom/kyant/**;
HSPLdev/chrisbanes/haze/**->**(**)**
Ldev/chrisbanes/haze/**;
# The player engine and the image loader: the first things touched when a song starts or a page of
# covers scrolls in. Neither ships a profile wide enough to cover the paths Exhale drives.
HSPLandroidx/media3/**->**(**)**
Landroidx/media3/**;
HSPLcoil3/**->**(**)**
Lcoil3/**;
HSPLokhttp3/**->**(**)**
Lokhttp3/**;
